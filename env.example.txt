# JWT Authentication Configuration
JWT_SECRET=your-jwt-secret-here
# Alternative: You can also use AUTH_SECRET (fallback if JWT_SECRET not set)

# Site URL (for OAuth redirects)
BETTER_AUTH_URL=http://localhost:4321
# Alternative: You can also use PUBLIC_SITE_URL (fallback if BETTER_AUTH_URL not set)

# Google OAuth Configuration with Calendar Access
# Get these from Google Cloud Console > Credentials > OAuth 2.0 Client IDs
# IMPORTANT: Enable Google Calendar API
GOOGLE_CLIENT_ID=your-google-client-id.apps.googleusercontent.com
GOOGLE_CLIENT_SECRET=your-google-client-secret

# IMPORTANT: In Google Cloud Console, set these exact URLs:
# Authorized JavaScript origins: http://localhost:4321
# Authorized redirect URIs: http://localhost:4321/api/auth/google-callback
# AND add calendar scope in OAuth consent screen!

# Optional: Restrict access to specific emails (comma-separated)
# Leave empty to allow all Google users
ALLOWED_EMAILS=your-email@example.com,another-email@example.com
